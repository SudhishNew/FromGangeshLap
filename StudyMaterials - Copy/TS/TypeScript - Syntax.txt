Inheritance Class:
class ParentClass {
    property: type;
    constructor(property: type) {
        this.property = property;
    }
    method() {
        console.log("Parent Method");
    }
}

class ChildClass extends ParentClass {
    childProperty: type;
    constructor(property: type, childProperty: type) {
        super(property); // Calls the parent constructor
        this.childProperty = childProperty;
    }
    childMethod() {
        console.log("Child Method");
    }
}
const myClass = new ParentClass (value1,value2)

Abstract Class:
abstract class ClassName {
    property: type;
    constructor(property: type) {
        this.property = property;
    }
    abstract methodName(): returnType;   // Abstract method (no implementation)
    display(): void {     // Concrete method (with implementation)
        console.log(this.property);
    }
}

class ChildClass extends ClassName {
    constructor(property: type) {
        super(property);
    }
    methodName(): returnType {   // Must implement the abstract method

    }
}
